// ===== ADMIN: ALL ENTRIES TABLE ===== async function loadAllEntries(){ const tbody = document.getElementById('adminTbody'); tbody.innerHTML = 'Loading…'; try{ const branchVal = document.getElementById('branchFilter').value; const dateVal = document.getElementById('dateFilter').value; // Index error se bachne ke liye sirf orderBy use kar rahe hain, filtering JS mein hogi const q = query(collection(db, CONFIG.ENTRIES_COLLECTION), orderBy('createdAt','desc')); const snap = await getDocs(q); let rows = snap.docs.map(d => ({ id: d.id, ...d.data() })); // JavaScript side filtering for branch and date if(branchVal){ rows = rows.filter(r => r.branch === branchVal); } if(dateVal){ rows = rows.filter(r => r.date === dateVal); } if(rows.length === 0){ tbody.innerHTML = 'No entries found.'; return; } tbody.innerHTML = rows.map(r => ` ${r.date} ${r.branch} ${r.tds} ${r.ph} ${r.temp} ${r.method} ${r.coffeeName} ${r.waterVolume} ${r.coffeeGrams} ${r.puls || ''} ${r.extractionTime || ''} ${r.extractionMl || ''} ${r.coffeeRatio || ''} `).join(''); document.querySelectorAll('.admin-del-btn').forEach(btn=>{ btn.addEventListener('click', async ()=>{ if(!confirm('Delete this entry? This cannot be undone.')) return; await deleteDoc(doc(db, CONFIG.ENTRIES_COLLECTION, btn.dataset.id)); showToast('Entry deleted'); loadAllEntries(); }); }); } catch(err){ tbody.innerHTML = 'Could not load data — check connection/setup.'; console.error(err); } }
